# Buildroot makeusers tablosu: kullanici uid grup gid parola ev kabuk gruplar aciklama
# 'makerd' = makerd HIZMET hesabi (giris yok; 1.0.1 ve oncesinde adi 'maker' idi, account-migrate cevirir).
# Insan kullanici (UID 1000, wheel; varsayilan 'maker')
# ilk acilis sihirbazinda makerd-priv user.create ile olusturulur.
makerd 999 makerd 999 * /var/lib/makerd /sbin/nologin video,gpio,i2c,spi,dialout,netdev,input,render,audio,bluetooth MakerOS hizmet hesabi (makerd)
- - wheel -1 * - - - Yoneticiler (sudo)
- - video -1 * - - - Video/kamera
- - gpio -1 * - - - GPIO erisimi
- - i2c -1 * - - - I2C erisimi
- - spi -1 * - - - SPI erisimi
- - dialout -1 * - - - Seri portlar
- - input -1 * - - - Giris aygitlari
- - render -1 * - - - DRI render dugumleri
- - audio -1 * - - - Ses
- - plugdev -1 * - - - Cikarilabilir aygitlar
- - netdev -1 * - - - Ag yonetimi
- - bluetooth -1 * - - - Bluetooth
